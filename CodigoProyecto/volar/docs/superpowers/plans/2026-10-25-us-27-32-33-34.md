**Goal:** Implement four user stories: US-27 (reportes de ocupación para el administrador), and US-32/US-33/US-34 (terminal de mostrador: búsqueda de vuelos, venta presencial y consulta de reservas para el empleado de mostrador).

## Decisiones de diseño (confirmar antes de implementar)

1. **US-33 es un único paso, no una copia del wizard web de 3 pasos.** El empleado y el pasajero están presentes en el momento: se busca el vuelo (US-32), se completa un solo formulario (cantidad implícita = cantidad de pasajeros cargados, clase, datos de cada pasajero, medio de cobro) y se confirma. Todo ocurre en una sola transacción atómica (reserva de cupo + reserva CONFIRMED + emisión de pasajes/factura), sin el hold de 10 minutos de la compra web (ese hold existe para cubrir los pasos intermedios de la compra online; aquí no hay pasos intermedios). Se reutiliza `issueComprobantes` tal cual.
2. **El email de contacto es opcional en mostrador.** El wireframe no lo pide, pero `Booking.contactEmail` es obligatorio en la base y RF-09 dice que los comprobantes se envían por correo. Si el empleado no lo completa, se guarda `""` y **no** se encola el email (se agrega una guarda mínima en `issuance.ts`). Si lo completa, se envía igual que en la compra web.
3. **US-27 usa "ocupados" (vendidos + retenidos) como "asientos vendidos",** la misma noción que ya usa toda la app (`economyOccupied`/`firstClassOccupied`, "cupo disponible = capacidad − ocupados"). No se introduce una métrica nueva basada solo en pasajes `ISSUED`.
4. **US-34 busca en TODAS las reservas (web y mostrador),** no solo las vendidas por mostrador: un empleado debe poder atender a cualquier pasajero, sin importar el canal de compra original.
5. **`/mostrador/*` es exclusivo del rol `COUNTER_AGENT`** (no lo ve un ADMIN), igual que `/admin/*` es exclusivo de `ADMIN` (US-30: "ver solo las funciones de mi rol").
6. **El reporte US-27 vive en `/admin/reportes`** y cubre solo la ocupación (US-27). La tabla de auditoría de pagos del mismo wireframe es US-22, que ya está implementado en otra parte del sistema (no se toca acá).
7. **La confirmación de una venta de mostrador reutiliza la pantalla de US-34**: al terminar la venta, se redirige a `/mostrador/reservas?codigo=<código>`, que muestra el detalle completo (mismos componentes que US-34). Evita construir una tercera pantalla de "comprobante" solo para mostrador.

// features/reports/schema.ts
import { z } from "zod";
import { DATE_REGEX, todayDateString } from "@/lib/dates";

/** Primer día del mes actual, formato "YYYY-MM-DD" (mismo criterio UTC que lib/dates). */
function firstDayOfCurrentMonth(): string {
  return `${todayDateString().slice(0, 7)}-01`;
}

export const REPORT_CLASS_FILTERS = ["todas", "ECONOMY", "FIRST"] as const;
export type ReportClassFilter = (typeof REPORT_CLASS_FILTERS)[number];

// US-27: filtros de la consulta de ocupación (reportes_admin.html). Por
// defecto, el mes en curso completo, todas las rutas y todas las clases.
export const reportFiltersSchema = z.object({
  desde: z.string().regex(DATE_REGEX).catch(firstDayOfCurrentMonth),
  hasta: z.string().regex(DATE_REGEX).catch(todayDateString),
  ruta: z.string().trim().catch(""),
  clase: z.enum(REPORT_CLASS_FILTERS).catch("todas"),
  page: z.coerce.number().int().min(1).catch(1),
});

export type ReportFilters = z.infer<typeof reportFiltersSchema>;
```

// features/reports/queries.ts
import { Prisma } from "@/generated/prisma/client";
import { prisma } from "@/lib/prisma";
import { parseDateString } from "@/lib/dates";
import { reportFiltersSchema, type ReportFilters } from "./schema";

export const REPORT_PAGE_SIZE = 10;

export type OccupancyReportRow = {
  id: string;
  code: string;
  date: Date;
  originCode: string;
  destinationCode: string;
  airplaneModel: string;
  economyCapacity: number;
  economySold: number;
  firstClassCapacity: number;
  firstClassSold: number;
  revenue: string;
};

export type OccupancySummary = {
  totalCapacity: number;
  totalSold: number;
  occupancyRate: number;
  economyOccupancyRate: number;
  firstClassOccupancyRate: number;
  grossRevenue: string;
};

export type OccupancyReport = {
  flights: OccupancyReportRow[];
  total: number;
  page: number;
  pageCount: number;
  filters: ReportFilters;
  summary: OccupancySummary;
};

function round1(value: number): number {
  return Math.round(value * 10) / 10;
}

function rate(sold: number, capacity: number): number {
  return capacity > 0 ? round1((sold / capacity) * 100) : 0;
}

/**
 * US-27: ocupación por vuelo, clase y fecha, con filtros por rango de fechas,
 * trayecto y clase, y KPIs globales (no solo de la página visible).
 * "Vendidos" = asientos ocupados (vendidos + retenidos), la misma noción de
 * cupo que usa el resto de la app (US-09/US-10: disponible = capacidad − ocupados).
 */
export async function getOccupancyReport(
  rawFilters: Record<string, string | undefined>,
): Promise<OccupancyReport> {
  const filters: ReportFilters = reportFiltersSchema.parse(rawFilters);

  const desde = parseDateString(filters.desde);
  const hastaParsed = parseDateString(filters.hasta);
  const hasta = hastaParsed < desde ? desde : hastaParsed;

  const where: Prisma.FlightWhereInput = {
    date: { gte: desde, lte: hasta },
    ...(filters.ruta && { routeId: filters.ruta }),
  };

  const [total, rows, capacityAgg, revenueAgg] = await Promise.all([
    prisma.flight.count({ where }),
    prisma.flight.findMany({
      where,
      orderBy: [{ date: "asc" }, { departureAt: "asc" }],
      skip: (filters.page - 1) * REPORT_PAGE_SIZE,
      take: REPORT_PAGE_SIZE,
      include: {
        route: { include: { origin: { select: { code: true } }, destination: { select: { code: true } } } },
        airplane: { select: { model: true } },
        tickets: { where: { status: "ISSUED" }, select: { unitPrice: true } },
      },
    }),
    prisma.flight.aggregate({
      where,
      _sum: {
        economyCapacity: true,
        economyOccupied: true,
        firstClassCapacity: true,
        firstClassOccupied: true,
      },
    }),
    prisma.ticket.aggregate({
      where: {
        status: "ISSUED",
        flight: where,
        ...(filters.clase !== "todas" && { seatClass: filters.clase }),
      },
      _sum: { unitPrice: true },
    }),
  ]);

  const flights: OccupancyReportRow[] = rows.map((f) => ({
    id: f.id,
    code: f.code,
    date: f.date,
    originCode: f.route.origin.code,
    destinationCode: f.route.destination.code,
    airplaneModel: f.airplane.model,
    economyCapacity: f.economyCapacity,
    economySold: f.economyOccupied,
    firstClassCapacity: f.firstClassCapacity,
    firstClassSold: f.firstClassOccupied,
    revenue: f.tickets.reduce((sum, t) => sum + Number(t.unitPrice), 0).toString(),
  }));

  const economyCapacity = capacityAgg._sum.economyCapacity ?? 0;
  const economyOccupied = capacityAgg._sum.economyOccupied ?? 0;
  const firstClassCapacity = capacityAgg._sum.firstClassCapacity ?? 0;
  const firstClassOccupied = capacityAgg._sum.firstClassOccupied ?? 0;

  const totalCapacity =
    filters.clase === "ECONOMY"
      ? economyCapacity
      : filters.clase === "FIRST"
        ? firstClassCapacity
        : economyCapacity + firstClassCapacity;
  const totalSold =
    filters.clase === "ECONOMY"
      ? economyOccupied
      : filters.clase === "FIRST"
        ? firstClassOccupied
        : economyOccupied + firstClassOccupied;

  const summary: OccupancySummary = {
    totalCapacity,
    totalSold,
    occupancyRate: rate(totalSold, totalCapacity),
    economyOccupancyRate: rate(economyOccupied, economyCapacity),
    firstClassOccupancyRate: rate(firstClassOccupied, firstClassCapacity),
    grossRevenue: (revenueAgg._sum.unitPrice ?? new Prisma.Decimal(0)).toString(),
  };

  return {
    flights,
    total,
    page: filters.page,
    pageCount: Math.max(1, Math.ceil(total / REPORT_PAGE_SIZE)),
    filters,
    summary,
  };
}
```

    return `/admin/reportes${qs ? `?${qs}` : ""}`;

// features/counter/service.ts
import { Prisma } from "@/generated/prisma/client";
import type { SeatClass } from "@/generated/prisma/enums";
import { prisma } from "@/lib/prisma";
import { newId } from "@/lib/id";
import type { ActionResult } from "@/lib/action-result";
import { availableSeats, isOnSale } from "@/features/flights/availability";
import { issueComprobantes } from "@/features/purchase/issuance";
import { newBookingCode, newTransactionId } from "@/features/purchase/service";
import type { PassengerValues } from "@/features/purchase/schema";
import type { CounterPaymentMethod } from "./schema";

type Seller = { id: string };

type CounterSaleInput = {
  flightId: string;
  seatClass: SeatClass;
  metodoPago: CounterPaymentMethod;
  contactEmail: string;
  pasajeros: PassengerValues[];
};

const COUNTER_PAYMENT_GATEWAY_LABEL: Record<CounterPaymentMethod, string> = {
  CASH: "Mostrador — Efectivo",
  POS: "Mostrador — POS",
};

/**
 * US-33: venta presencial en mostrador. A diferencia de la compra web
 * (US-15/US-17/US-20/US-21), no hay hold de 10 minutos ni pasarela que pueda
 * rechazar el pago: el empleado y el pasajero están presentes y el cobro ya
 * ocurrió físicamente. Reserva el cupo de forma atómica (misma técnica que
 * `createBooking`), crea la reserva ya CONFIRMED, registra el pago y emite
 * pasajes y factura con `issueComprobantes` (US-23/US-24), todo en una sola
 * transacción.
 */
export async function createCounterSale(
  input: CounterSaleInput,
  seller: Seller,
): Promise<ActionResult<{ code: string }>> {
  const quantity = input.pasajeros.length;
  const now = new Date();

  const flight = await prisma.flight.findUnique({
    where: { id: input.flightId },
    include: { salesPeriod: { select: { startDate: true, endDate: true } } },
  });
  if (!flight || !isOnSale(flight, now)) {
    return { ok: false, error: "Este vuelo ya no está disponible para la venta." };
  }

  const capacity = input.seatClass === "ECONOMY" ? flight.economyCapacity : flight.firstClassCapacity;
  const occupied = input.seatClass === "ECONOMY" ? flight.economyOccupied : flight.firstClassOccupied;
  const fare = input.seatClass === "ECONOMY" ? flight.economyFare : flight.firstClassFare;

  const available = availableSeats(capacity, occupied);
  if (quantity > available) {
    return {
      ok: false,
      error:
        available === 0
          ? "No quedan cupos disponibles en esta clase."
          : `Solo quedan ${available} cupo(s) disponibles en esta clase.`,
      fieldErrors: { pasajeros: [`Quedan ${available} cupo(s) disponibles`] },
    };
  }

  const bookingId = newId("Booking");
  const code = newBookingCode();
  const totalAmount = new Prisma.Decimal(fare).times(quantity);
  const maxOccupiedAllowed = capacity - quantity;
  const titular = input.pasajeros[0];
  const contactName = `${titular.apellido}, ${titular.nombre}`;

  try {
    await prisma.$transaction(async (tx) => {
      const { count } =
        input.seatClass === "ECONOMY"
          ? await tx.flight.updateMany({
              where: { id: flight.id, economyOccupied: { lte: maxOccupiedAllowed } },
              data: { economyOccupied: { increment: quantity } },
            })
          : await tx.flight.updateMany({
              where: { id: flight.id, firstClassOccupied: { lte: maxOccupiedAllowed } },
              data: { firstClassOccupied: { increment: quantity } },
            });
      if (count === 0) throw new Error("NO_SEATS");

      await tx.booking.create({
        data: {
          id: bookingId,
          code,
          flightId: flight.id,
          seatClass: input.seatClass,
          quantity,
          unitPrice: fare,
          totalAmount,
          status: "CONFIRMED",
          holdExpiresAt: now,
          channel: "COUNTER",
          sellerId: seller.id,
          contactEmail: input.contactEmail,
          contactName,
          confirmedAt: now,
        },
      });

      const tickets = input.pasajeros.map((p) => ({
        id: newId("Ticket"),
        bookingId,
        flightId: flight.id,
        seatClass: input.seatClass,
        unitPrice: fare,
        firstName: p.nombre,
        lastName: p.apellido,
        documentType: p.tipoDocumento,
        documentNumber: p.numeroDocumento,
        status: "PENDING" as const,
      }));
      await tx.ticket.createMany({ data: tickets });

      await tx.payment.create({
        data: {
          id: newId("Payment"),
          bookingId,
          amount: totalAmount,
          status: "APPROVED",
          gateway: COUNTER_PAYMENT_GATEWAY_LABEL[input.metodoPago],
          transactionId: newTransactionId(),
          processedAt: now,
        },
      });

      await issueComprobantes(
        tx,
        {
          id: bookingId,
          code,
          flightId: flight.id,
          seatClass: input.seatClass,
          contactName,
          contactEmail: input.contactEmail,
          totalAmount,
          flight: {
            code: flight.code,
            economyCapacity: flight.economyCapacity,
            firstClassCapacity: flight.firstClassCapacity,
          },
          tickets: tickets.map((t) => ({
            id: t.id,
            firstName: t.firstName,
            lastName: t.lastName,
            unitPrice: t.unitPrice,
          })),
        },
        now,
      );
    });
  } catch (err) {
    if (err instanceof Error && err.message === "NO_SEATS") {
      return {
        ok: false,
        error: "Se agotaron los cupos de esta clase justo ahora. Ajustá la cantidad o elegí otra opción.",
      };
    }
    if (err instanceof Prisma.PrismaClientKnownRequestError && err.code === "P2002") {
      return { ok: false, error: "No se pudo registrar la venta. Volvé a intentarlo." };
    }
    throw err;
  }

  return { ok: true, data: { code } };
}

"use server";

import { prisma } from "@/lib/prisma";
import { requireCounterAgent } from "@/lib/auth";
import { sendPurchaseConfirmationEmail } from "@/lib/mailer/purchase";
import type { ActionResult } from "@/lib/action-result";
import { counterSaleSchema, type CounterSaleValues } from "./schema";
import { createCounterSale } from "./service";

/**
 * US-33: registra la venta presencial. Si se indicó email de contacto, envía
 * los comprobantes (igual que `processPaymentAction` en la compra web): el
 * envío es best-effort y nunca convierte una venta exitosa en un error.
 */
export async function createCounterSaleAction(
  values: CounterSaleValues,
): Promise<ActionResult<{ code: string }>> {
  const agent = await requireCounterAgent();

  const parsed = counterSaleSchema.safeParse(values);
  if (!parsed.success) {
    const fieldErrors: Record<string, string[]> = {};
    for (const issue of parsed.error.issues) {
      (fieldErrors[issue.path.join(".")] ??= []).push(issue.message);
    }
    return { ok: false, error: "Revisá los datos de la venta.", fieldErrors };
  }

  const result = await createCounterSale(parsed.data, agent);

  if (result.ok && parsed.data.contactEmail) {
    try {
      const notification = await prisma.emailNotification.findFirst({
        where: {
          type: "PURCHASE_CONFIRMATION",
          status: "PENDING",
          booking: { code: result.data.code },
        },
        orderBy: { createdAt: "desc" },
      });
      if (notification) await sendPurchaseConfirmationEmail(notification, result.data.code);
    } catch (error) {
      console.error("No se pudo enviar la confirmación de la venta de mostrador:", error);
    }
  }

  return result;
}

// features/counter/queries.ts
import { prisma } from "@/lib/prisma";
import type { BookingStatus, SalesChannel } from "@/generated/prisma/enums";

export type ReservationSummary = {
  code: string;
  contactName: string;
  status: BookingStatus;
  channel: SalesChannel;
  flightCode: string;
  originCode: string;
  destinationCode: string;
  date: Date;
};

/**
 * US-34: busca reservas (de cualquier canal, web o mostrador) por código
 * exacto, por el nombre de contacto de la compra, o por el documento/apellido
 * de alguno de sus pasajeros. No expone datos de pago (ver `getBookingComprobantes`,
 * que tampoco los selecciona: cumple RNF-08).
 */
export async function searchReservations(criterio: string): Promise<ReservationSummary[]> {
  const value = criterio.trim();
  if (!value) return [];

  const rows = await prisma.booking.findMany({
    where: {
      OR: [
        { code: value.toUpperCase() },
        { contactName: { contains: value, mode: "insensitive" } },
        {
          tickets: {
            some: {
              OR: [
                { documentNumber: value },
                { lastName: { contains: value, mode: "insensitive" } },
              ],
            },
          },
        },
      ],
    },
    orderBy: { createdAt: "desc" },
    take: 20,
    include: {
      flight: {
        select: {
          code: true,
          date: true,
          route: {
            select: { origin: { select: { code: true } }, destination: { select: { code: true } } },
          },
        },
      },
    },
  });

  return rows.map((b) => ({
    code: b.code,
    contactName: b.contactName,
    status: b.status,
    channel: b.channel,
    flightCode: b.flight.code,
    originCode: b.flight.route.origin.code,
    destinationCode: b.flight.route.destination.code,
    date: b.flight.date,
  }));
}

  const sp = await searchParams;
  const parsed = sp.criterio !== undefined ? reservationSearchSchema.safeParse(sp) : null;
  const results = parsed?.success ? await searchReservations(parsed.data.criterio) : [];

  const selectedCode = typeof sp.codigo === "string" ? sp.codigo.trim().toUpperCase() : null;
  const detail = selectedCode ? await getBookingComprobantes(selectedCode) : null;

  return (
    <div className="space-y-6">
      <ReservationSearchForm defaultValue={parsed?.success ? parsed.data.criterio : ""} />

      {parsed && !parsed.success && (
        <p className="text-sm text-destructive">{parsed.error.issues[0]?.message}</p>
      )}

      {parsed?.success && !detail && (
        <ReservationResultsList criterio={parsed.data.criterio} results={results} />
      )}

      {detail && <ReservationDetail booking={detail} />}
    </div>
  );
}
